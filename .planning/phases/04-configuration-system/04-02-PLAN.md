```go
package setup

import (
    "fmt"
    "os"
    "path/filepath"

    cp "github.com/otiai10/copy"
)

// CopyPath copies a file or directory from src to dest, creating parent dirs as needed
func CopyPath(src, dest string) error {
    // Ensure parent directory exists
    if err := os.MkdirAll(filepath.Dir(dest), 0755); err != nil {
        return fmt.Errorf("failed to create parent directory: %w", err)
    }

    if err := cp.Copy(src, dest); err != nil {
        return fmt.Errorf("failed to copy %s: %w", src, err)
    }
    return nil
}
```

```go
package setup

// RunCommand executes a command via sh -c with the given working directory
// Stdout and stderr are streamed to the user in real-time
func RunCommand(workDir, command string) error {
    cmd := exec.Command("sh", "-c", command)
    cmd.Dir = workDir
    cmd.Stdout = os.Stdout
    cmd.Stderr = os.Stderr

    if err := cmd.Run(); err != nil {
        if exitErr, ok := err.(*exec.ExitError); ok {
            return fmt.Errorf("command failed with exit code %d", exitErr.ExitCode())
        }
        return fmt.Errorf("command failed: %w", err)
    }
    return nil
}
```

For symlinks: the `src` parameter must be an absolute path so the symlink target resolves correctly regardless of working directory. Use `filepath.Abs(src)` if needed, or ensure caller passes absolute paths.

```go
package setup

import (
    "fmt"
    "os"
    "os/exec"
    "path/filepath"

    "github.com/ahmedelarabyy/wt/internal/config"
)

// ExecuteActions runs all actions sequentially against the target worktree.
// srcRoot is the source worktree (where files are copied/symlinked from).
// targetRoot is the new worktree (where files land and commands run).
// If any action fails, the entire worktree is rolled back.
func ExecuteActions(srcRoot, targetRoot string, actions []config.Action) error {
    if len(actions) == 0 {
        return nil
    }

    for _, action := range actions {
        if err := executeOne(srcRoot, targetRoot, action); err != nil {
            // Rollback: remove entire worktree
            rollbackWorktree(targetRoot)
            return err
        }
    }

    return nil
}
```

The `executeOne` function dispatches based on action type:
- `config.ActionCopy`:
  - src = `filepath.Join(srcRoot, action.Path)`
  - dest = `filepath.Join(targetRoot, action.Path)`
  - Call `CopyPath(src, dest)`
  - Print `fmt.Printf("Copied %s\n", action.Path)`
- `config.ActionSymlink`:
  - src = `filepath.Join(srcRoot, action.Path)` — use absolute path
  - dest = `filepath.Join(targetRoot, action.Path)`
  - Call `CreateSymlink(src, dest)`
  - Print `fmt.Printf("Symlinked %s\n", action.Path)`
- `config.ActionRun`:
  - Print `fmt.Printf("Running %s...\n", action.Path)` (Path holds the command for run actions)
  - Call `RunCommand(targetRoot, action.Path)`

The `rollbackWorktree` function:
- Print `fmt.Fprintf(os.Stderr, "Error occurred, rolling back worktree...\n")`
- Run `git worktree remove --force <targetRoot>`:
  ```go
  removeCmd := exec.Command("git", "worktree", "remove", "--force", targetRoot)
  if err := removeCmd.Run(); err != nil {
      fmt.Fprintf(os.Stderr, "WARNING: rollback failed: %v\n", err)
      fmt.Fprintf(os.Stderr, "Manual cleanup: git worktree remove --force %s\n", targetRoot)
      return
  }
  ```
- Clean up directory if git left it:
  ```go
  if _, err := os.Stat(targetRoot); err == nil {
      if err := os.RemoveAll(targetRoot); err != nil {
          fmt.Fprintf(os.Stderr, "WARNING: directory cleanup failed: %v\n", err)
          fmt.Fprintf(os.Stderr, "Manual cleanup: rm -rf %s\n", targetRoot)
      }
  }
  ```
